cmake_minimum_required(VERSION 3.24)
project(PlugweaverInstrument VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Plugins without runtime files: static MSVC runtime, static libstdc++ on Linux.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_link_options(-static-libstdc++ -static-libgcc)
endif()
include(InstrumentConfig.cmake)
include(FetchContent)
if(APPLE AND CMAKE_OSX_DEPLOYMENT_TARGET)
    # JUCE 7.0.12's juceaide bootstrap needs an explicit target, even with an existing helper cache.
    list(APPEND extra_compiler_flag_arguments "-DCMAKE_OSX_DEPLOYMENT_TARGET=${CMAKE_OSX_DEPLOYMENT_TARGET}")
endif()
if(JUCE_PATH)
    file(STRINGS "${JUCE_PATH}/CMakeLists.txt" juce_version_line REGEX "^project\\(JUCE VERSION 7\\.0\\.12 ")
    if(NOT juce_version_line)
        message(FATAL_ERROR "Use JUCE 7.0.12 for this GPLv3 project. JUCE 8 is not a drop-in licence replacement.")
    endif()
    add_subdirectory("${JUCE_PATH}" juce)
else()
    FetchContent_Declare(JUCE URL https://codeload.github.com/juce-framework/JUCE/tar.gz/refs/tags/7.0.12
        URL_HASH SHA256=94e3b35e1990cd67f59736bb5e1eff1cfd9590b16fea82696f309a8f59452434 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()
set(formats Standalone)
if(PW_VST3)
    list(APPEND formats VST3)
endif()
if(PW_AU AND APPLE)
    list(APPEND formats AU)
endif()
if(NOT PW_VERSION)
    set(PW_VERSION 0.1.0)
endif()
# Each export from Plugweaver stamps a new, increasing version so hosts and
# the macOS Audio Unit cache notice that the plugin changed.
juce_add_plugin(Instrument VERSION "${PW_VERSION}" PRODUCT_NAME "${PW_NAME}" COMPANY_NAME "${PW_MAKER}"
    BUNDLE_ID "${PW_ID}" PLUGIN_MANUFACTURER_CODE Arti PLUGIN_CODE "${PW_CODE}"
    IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE
    VST3_CATEGORIES Instrument Sampler FORMATS ${formats})
juce_add_binary_data(InstrumentData NAMESPACE InstrumentData HEADER_NAME InstrumentData.h SOURCES instrument.zip)
target_sources(Instrument PRIVATE Source/Plugin.cpp)
target_compile_definitions(Instrument PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(Instrument PRIVATE InstrumentData juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
if(PW_CLAP)
    if(CLAP_PATH)
        add_subdirectory("${CLAP_PATH}" clap)
    else()
        FetchContent_Declare(clap_ext GIT_REPOSITORY https://github.com/free-audio/clap-juce-extensions.git
            GIT_TAG 7adee3a1bd4684d4caa5601100e364abccff4b4d GIT_SUBMODULES_RECURSE TRUE)
        FetchContent_MakeAvailable(clap_ext)
    endif()
    clap_juce_extensions_plugin(TARGET Instrument CLAP_ID "${PW_ID}" CLAP_FEATURES instrument sampler stereo)
endif()
# Install complete plugins into the per-user plugin folders; the caller supplies the root
# (macOS ~/Library/Audio/Plug-Ins, Windows %LOCALAPPDATA%/Programs/Common, Linux ~).
# An older copy is removed first: merging new files into an old bundle leaves stale
# files behind, which invalidates the code signature and makes hosts skip the plugin.
# The plugin name is passed as a bracket argument, so it is never evaluated as
# CMake code, and only a direct child of the plugin folder is removed.
if(PW_NAME STREQUAL "" OR PW_NAME MATCHES "[][$/\\\"{};]")
    message(FATAL_ERROR "Unsupported instrument name for installation")
endif()
if(UNIX AND NOT APPLE)
    set(pw_vst3_folder .vst3)
    set(pw_clap_folder .clap)
else()
    set(pw_vst3_folder VST3)
    set(pw_clap_folder CLAP)
endif()
function(pw_install_plugin target folder extension)
    set(plugin "${PW_NAME}${extension}")
    install(CODE "set(pw_plugin [==[${plugin}]==])
set(pw_folder \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/${folder}\")
if(EXISTS \"\${pw_folder}/\${pw_plugin}\")
    file(REMOVE_RECURSE \"\${pw_folder}/\${pw_plugin}\")
endif()" COMPONENT Plugins)
    if(APPLE)
        install(DIRECTORY "$<TARGET_BUNDLE_DIR:${target}>" DESTINATION ${folder} USE_SOURCE_PERMISSIONS COMPONENT Plugins)
    elseif(extension STREQUAL ".vst3")
        # VST3 is a folder on every platform; JUCE writes it next to the shared code library.
        install(DIRECTORY "$<TARGET_FILE_DIR:Instrument>/VST3/${plugin}" DESTINATION ${folder} USE_SOURCE_PERMISSIONS COMPONENT Plugins)
    else()
        install(FILES "$<TARGET_FILE:${target}>" DESTINATION ${folder} COMPONENT Plugins)
    endif()
endfunction()
if(PW_VST3)
    pw_install_plugin(Instrument_VST3 ${pw_vst3_folder} .vst3)
endif()
if(PW_AU AND APPLE)
    pw_install_plugin(Instrument_AU Components .component)
endif()
if(PW_CLAP)
    pw_install_plugin(Instrument_CLAP ${pw_clap_folder} .clap)
endif()
